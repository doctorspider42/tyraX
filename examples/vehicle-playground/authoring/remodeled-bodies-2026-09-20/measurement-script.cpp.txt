static void captureVehicleFrame(Vehicle_playground::ScriptContext& ctx) {
 auto* fb=ctx.engine->renderer.core.gs.getPreviousRealFrameBuffer();
 if(!fb || fb->width>1024) return;
 FILE* f=std::fopen(Tyra::FileUtils::fromCwd("vehicle-capture.tga").c_str(),"wb");
 if(!f)return;
 unsigned char hdr[18]={};hdr[2]=2;hdr[12]=fb->width&255;hdr[13]=fb->width>>8;
 hdr[14]=fb->height&255;hdr[15]=fb->height>>8;hdr[16]=32;hdr[17]=8;
 std::fwrite(hdr,1,18,f);
 static unsigned int in[1024] __attribute__((aligned(16))),out[1024];
 for(unsigned y=0;y<fb->height;y++){
  if(!ps2_screenshot(in,fb->address/64,0,fb->height-1-y,fb->width,1,fb->psm)){std::fclose(f);return;}
  FlushCache(0);
  for(unsigned x=0;x<fb->width;x++){
   unsigned r,g,b;
   if(fb->psm==2){unsigned c=((unsigned short*)in)[x];r=(c&31)<<3;g=((c>>5)&31)<<3;b=((c>>10)&31)<<3;}
   else{auto* c=((unsigned char*)in)+x*(fb->psm==1?3:4);r=c[0];g=c[1];b=c[2];}
   out[x]=0xff000000|(r<<16)|(g<<8)|b;
  }
  std::fwrite(out,4,fb->width,f);
 }
 std::fclose(f);
}

namespace Vehicle_playground {
class DistrictBenchmark : public Script {
  unsigned int frame = 0;
  struct Sample { unsigned int frame, phase; float fps; } samples[32];
  int count = 0;
  unsigned int heldPhase = 3;
  bool written = false;
  unsigned capturedPose=99;
 public:
  void update(ScriptContext& ctx) override {
    // Read pose commands only AFTER measurement; captures must not disturb samples.
    if (frame >= 1440 && frame % 30 == 0) {
      FILE* f = std::fopen(Tyra::FileUtils::fromCwd("district-benchmark-pose.txt").c_str(), "r");
      if (f) {
        unsigned int requested = 3;
        if (std::fscanf(f, "%u", &requested) == 1 && requested < 6)
          heldPhase = requested;
        std::fclose(f);
      }
    }
    const unsigned int phase = frame < 1440 ? frame / 360 : heldPhase;
    if (ctx.saveValues && DISTRICT_NIGHT_VALUE < ctx.saveValueCount)
      ctx.saveValues[DISTRICT_NIGHT_VALUE] = (phase & 1) ? 1.0F : 0.0F;
    ctx.cameraOverride = true;
    ctx.cameraEye = phase < 2 ? Tyra::Vec4(0,4,-32,1) : Tyra::Vec4(4,9,102,1);
    ctx.cameraAt = phase < 2 ? Tyra::Vec4(0,1,-12,1) : Tyra::Vec4(65,3,106,1);
    ctx.cameraUp = Tyra::Vec4(0,1,0,0);
    if (phase == 4) { ctx.cameraEye=Tyra::Vec4(-8,3,-18,1); ctx.cameraAt=Tyra::Vec4(-2.14626F,1,-11.5592F,1); }
    if (phase == 5) { ctx.cameraEye=Tyra::Vec4(-13,2.4F,-27,1); ctx.cameraAt=Tyra::Vec4(-8,.85F,-22,1); if(ctx.saveValues) ctx.saveValues[DISTRICT_NIGHT_VALUE]=0; }
    if (frame < 1440 && frame % 360 >= 120 && frame % 30 == 0 && count < 32)
      samples[count++] = {frame, phase, ctx.engine->info.getFps()};
    if (frame >= 1440 && !written) {
      FILE* f = std::fopen(Tyra::FileUtils::fromCwd("district-benchmark.csv").c_str(), "w");
      if (f) {
        std::fprintf(f,"frame,phase,fps\n");
        for (int i=0;i<count;++i)
          std::fprintf(f,"%u,%u,%.3f\n",samples[i].frame,samples[i].phase,samples[i].fps);
        std::fclose(f);
        written = true;
      }
    }
    if(frame>1470 && frame%30==0 && capturedPose!=heldPhase) {
      captureVehicleFrame(ctx);capturedPose=heldPhase;
    }
    ++frame;
  }
};
}
TYRA_SCRIPT(Vehicle_playground::DistrictBenchmark);
